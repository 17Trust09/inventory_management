🔧 ESP32 Regal-LED Setup

So richtest du den ESP32 ein, der markierte Lagerorte per LED anzeigt.

🔄 Funktionsweise
  1. In der Lagerdatenbank wird ein Item markiert (Button Markieren)
  2. Django ermittelt den untersten StorageLocation
  3. Eintrag in ItemMark-Tabelle mit location_id
  4. ESP pollt alle 3s die API
  5. API liefert location_id jedes markierten Items
  6. ESP mapped location_id auf LED und leuchtet sie

const int   LED_TIMEOUT_SEC  = 300;

#endif
📄 src/position_map.h – Schubladen auf LEDs mappen Hier anpassen! 
#ifndef POSITION_MAP_H
#define POSITION_MAP_H

struct PositionEntry {
    int locationId;
    int ledIndex;
};

// IDs findest du: Admin -> Lagerorte, oder via API
const PositionEntry POSITION_MAP[] = {
    {1,  0},   // Schublade 1 -> LED #0
    {2,  1},   // Schublade 2 -> LED #1
    {3,  2},   // Schublade 3 -> LED #2
    {4,  3},   // Schublade 4 -> LED #3
    {5,  4},   // Schublade 5 -> LED #4
    {6,  5},   // Schublade 6 -> LED #5
    {7,  6},   // Schublade 7 -> LED #6
    {8,  7},   // Schublade 8 -> LED #7
    {9,  8},   // Schublade 9 -> LED #8
    {10, 9},  // Schublade 10 -> LED #9
};

const size_t POSITION_COUNT = sizeof(POSITION_MAP) / sizeof(POSITION_MAP[0]);

#endif
📄 src/regal_led.ino – Hauptprogramm muss nicht angepasst werden 
#include <Arduino.h>
#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <Adafruit_NeoPixel.h>
#include "config.h"
#include "position_map.h"

Adafruit_NeoPixel strip(NUM_LEDS, LED_PIN, NEO_GRB + NEO_KHZ800);
bool ledState[NUM_LEDS] = {false};
unsigned long ledTimestamps[NUM_LEDS] = {0};

void connectWiFi() {
    Serial.print("Verbinde mit WLAN: ");
    Serial.println(WLAN_SSID);
    WiFi.mode(WIFI_STA);
    WiFi.begin(WLAN_SSID, WLAN_PASS);
    int a = 0;
    while (WiFi.status() != WL_CONNECTED && a < 40) {
        delay(500); Serial.print("."); a++;
    }
    if (WiFi.status() == WL_CONNECTED) {
        Serial.println(); Serial.print("IP: "); Serial.println(WiFi.localIP());
    } else {
        Serial.println("\nFehlgeschlagen. Neustart...");
        delay(30000); ESP.restart();
    }
}

bool fetchMarkedItems(JsonDocument &doc) {
    HTTPClient http;
    String url = String(API_URL) + "?key=" + String(API_KEY);
    http.begin(url); http.setTimeout(5000);
    int code = http.GET();
    if (code != 200) { http.end(); return false; }
    DeserializationError err = deserializeJson(doc, http.getStream());
    if (err) { http.end(); return false; }
    http.end(); return true;
}

int locationIdToLed(int locationId) {
    for (size_t i = 0; i < POSITION_COUNT; i++)
        if (POSITION_MAP[i].locationId == locationId) return POSITION_MAP[i].ledIndex;
    return -1;
}

void updateLEDs() {
    StaticJsonDocument<16384> doc;
    if (!fetchMarkedItems(doc)) { Serial.println("API nicht erreichbar"); return; }
    JsonArray marks = doc["marks"].as<JsonArray>();
    bool newState[NUM_LEDS] = {false};
    unsigned long now = millis();
    for (JsonObject m : marks) {
        int loc = m["location_id"] | 0;
        if (loc == 0) continue;
        int led = locationIdToLed(loc);
        if (led >= 0 && led < NUM_LEDS) { newState[led] = true; ledTimestamps[led] = now; }
    }
    for (int i = 0; i < NUM_LEDS; i++) {
        if (newState[i] != ledState[i]) {
            ledState[i] = newState[i];
            strip.setPixelColor(i, newState[i] ? MARK_COLOR : 0);
        } else if (newState[i] && (now - ledTimestamps[i] > LED_TIMEOUT_SEC * 1000UL)) {
            ledState[i] = false; strip.setPixelColor(i, 0);
        }
    }
    strip.show();
}

void setup() {
    Serial.begin(115200); delay(500);
    Serial.println("\nESP32 Regal-LED v1.0");
    strip.begin(); strip.setBrightness(BRIGHTNESS); strip.show();
    connectWiFi();
}

void loop() {
    if (WiFi.status() != WL_CONNECTED) connectWiFi();
    updateLEDs();
    delay(POLL_INTERVAL_MS);
}

Mit PlatformIO (empfohlen)
  1. VS Code + PlatformIO Extension installieren
  2. Ordner esp32-led-marker/ oeffnen
  3. config.h und position_map.h anpassen
  4. ESP per USB anschliessen
  5. Auf Upload and Monitor klicken
Mit Arduino IDE
  1. regal_led.ino oeffnen
  2. .h-Dateien muessen im gleichen Ordner liegen
  3. Board: ESP32 Dev Module
  4. Bibliotheken: Adafruit NeoPixel, ArduinoJson
  5. Sketch -> Upload

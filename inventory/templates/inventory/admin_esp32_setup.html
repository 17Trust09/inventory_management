🔄 Funktionsweise
  1. In der Lagerdatenbank wird ein Item markiert (Button Markieren)
  2. Django ermittelt den untersten StorageLocation
  3. Eintrag in ItemMark-Tabelle mit location_id
  4. ESP pollt alle 3s die API
  5. API liefert location_id jedes markierten Items
  6. ESP mapped location_id auf LED und leuchtet sie
  7. Nach der eingestellten Sekundenzahl wird die Markierung automatisch aufgehoben

📄 src/position_map.h – Schubladen auf LEDs mappen Hier anpassen! 
#ifndef POSITION_MAP_H
#define POSITION_MAP_H

struct PositionEntry {
    int locationId;
    int ledIndex;
};

// IDs findest du: Admin -> Lagerorte, oder via API
const PositionEntry POSITION_MAP[] = {
    {5,  0},   // Schublade 1 -> LED #0
    {6,  1},   // Schublade 2 -> LED #1
    {7,  2},   // Schublade 3 -> LED #2
    {8,  3},   // Schublade 4 -> LED #3
    {9,  4},   // Schublade 5 -> LED #4
    {10, 5},   // Schublade 6 -> LED #5
    {11, 6},   // Schublade 7 -> LED #6
    {12, 7},   // Schublade 8 -> LED #7
};

const size_t POSITION_COUNT = sizeof(POSITION_MAP) / sizeof(POSITION_MAP[0]);

#endif
📄 src/regal_led.ino – Hauptprogramm muss nicht angepasst werden 
#include <Arduino.h>
#include <WiFi.h>
#include <HTTPClient.h>
#include <ArduinoJson.h>
#include <Adafruit_NeoPixel.h>
#include "config.h"
#include "position_map.h"

Adafruit_NeoPixel strip(NUM_LEDS, LED_PIN, NEO_GRB + NEO_KHZ800);
bool ledState[NUM_LEDS] = {false};

void connectWiFi() {
    Serial.println();
    Serial.print("Verbinde mit WLAN: ");
    Serial.println(WLAN_SSID);
    WiFi.mode(WIFI_STA);
    WiFi.begin(WLAN_SSID, WLAN_PASS);
    int attempts = 0;
    while (WiFi.status() != WL_CONNECTED && attempts < 40) {
        delay(500); Serial.print("."); attempts++;
    }
    if (WiFi.status() == WL_CONNECTED) {
        Serial.println(); Serial.print("IP: "); Serial.println(WiFi.localIP());
    } else {
        Serial.println(); Serial.println("WLAN-Fehlgeschlagen. Neustart...");
        delay(30000); ESP.restart();
    }
}

bool fetchMarkedItems(DynamicJsonDocument &doc) {
    HTTPClient http;
    if (WiFi.status() != WL_CONNECTED) return false;
    String url = String(API_URL) + "?key=" + String(API_KEY);
    http.begin(url); http.setTimeout(5000);
    int code = http.GET();
    if (code != 200) { http.end(); return false; }
    DeserializationError err = deserializeJson(doc, http.getStream());
    if (err) { http.end(); return false; }
    http.end(); return true;
}

int locationIdToLed(int locationId) {
    for (size_t i = 0; i < POSITION_COUNT; i++)
        if (POSITION_MAP[i].locationId == locationId) return POSITION_MAP[i].ledIndex;
    return -1;
}

void updateLEDs() {
    Serial.println("Frage API ab...");
    DynamicJsonDocument doc(8192);
    if (!fetchMarkedItems(doc)) { Serial.println("API nicht erreichbar."); return; }
    JsonArray marks = doc["marks"].as();
    int count = doc["count"].as();
    Serial.print("API Antwort: "); Serial.print(count); Serial.println(" markierte Items");
    for (int i = 0; i < NUM_LEDS; i++) {
        if (ledState[i]) { ledState[i] = false; strip.setPixelColor(i, 0); }
    }
    for (JsonObject mark : marks) {
        int locationId = mark["location_id"] | 0;
        if (locationId == 0) continue;
        int ledIdx = locationIdToLed(locationId);
        if (ledIdx >= 0 && ledIdx < NUM_LEDS) {
            Serial.printf("  -> Location-ID %d -> LED %d\n", locationId, ledIdx);
            ledState[ledIdx] = true;
            strip.setPixelColor(ledIdx, MARK_COLOR);
        } else {
            Serial.printf("  -> Location-ID %d (%s) nicht in POSITION_MAP\n", locationId, mark["name"] | "?");
        }
    }
    strip.show();
}

void setup() {
    Serial.begin(115200); delay(500);
    Serial.println(); Serial.println("ESP32 Regal-LED v1.0");
    strip.begin(); strip.setBrightness(BRIGHTNESS); strip.show();
    Serial.printf("LEDs: %d (Pin %d)\n", NUM_LEDS, LED_PIN);
    connectWiFi();
    updateLEDs();
}

void loop() {
    delay(POLL_INTERVAL_MS);
    updateLEDs();
}
